## std::thread::experimental::threadgroup(<Task, Result>)
  def ThreadTaskFn = fn Result(Task task, void* context)
  def ThreadInitFn = fn void(void* init_context)
  def ThreadFinalizerFn = fn void(void* finalizer_context, void* init_context)
  def ThreadGroupOutChannel = void
  // Lifecycle state of a ThreadGroup.
  enum ThreadGroupState(<Task, Result>)
    NO_INIT
    INIT
    RUNNING
    STOPPING
  // The outcome of one task. The `task` field holds the input that was submitted,
  struct ThreadTaskResult(<Task, Result>)
    .task: Task
    .result: Result
  // ThreadGroup is a worker-pool abstraction. Submit Tasks via add_task; the workers
  struct ThreadGroup(<Task, Result>)
    .task_fn: ThreadTaskFn
    .context: void*
    .init_fn: ThreadInitFn
    .init_context: void*
    .finalizer_fn: ThreadFinalizerFn
    .finalizer_context: void*
    .allocator: Allocator
    .mx: Mutex
    .done: ConditionVariable
    .?: void
    .outbound: ThreadGroupOutChannel*
    .threads: Thread[]
    .pending: Atomic{sz}
    .thread_settings: ThreadSettings
    .queue_type: ThreadGroupQueueType
    .state: ThreadGroupState
    // Add a task. If the queue is full, block.
    fn ThreadGroup.add_task(<Task, Result>)(Task task) -> void
    // Add a task. If the queue is full, run the task on the current thread instead.
    fn ThreadGroup.add_task_or_run(<Task, Result>)(Task task) -> void
    // Wait for all pending tasks to finish and return the tasks + result as a list.
    fn ThreadGroup.collect(<Task, Result>)(Allocator allocator, bool caller_help_finish = false) -> List{ThreadTaskResult}
    // Wait for all pending tasks to finish and return the values as a List.
    fn ThreadGroup.collect_values(<Task, Result>)(Allocator allocator, bool caller_help_finish = false) -> List{Result}
    // Finish any pending tasks, then clean up all resources. Any results not yet consumed are discarded.
    fn ThreadGroup.destroy(<Task, Result>)(bool caller_help_finish = false) -> void
    // Collect the tasks + result in a list, without waiting to ensure that all pending tasks are completed.
    fn ThreadGroup.drain(<Task, Result>)(Allocator allocator) -> List{ThreadTaskResult}
    // Collect all available results in a list, without waiting for all pending tasks to complete.
    fn ThreadGroup.drain_values(<Task, Result>)(Allocator allocator) -> List{Result}
    // Discard any results that have been produced but not yet consumed via pop_result/drain.
    fn ThreadGroup.drop_results(<Task, Result>)() -> void
    // Initialize a ThreadGroup. The group must be started with start() before tasks
    fn ThreadGroup.init(<Task, Result>)(Allocator allocator, ThreadTaskFn task_fn, int thread_count = 0, sz queue_capacity = 0, void* context = null, ThreadSettings thread_settings = {}) -> void?
    // Wait for all pending tasks to finish.
    fn ThreadGroup.join(<Task, Result>)(bool caller_help_finish = false) -> void
    // Pop the task + result, if available – blocking if there is no result available yet. Returns an excuse
    fn ThreadGroup.pop_result(<Task, Result>)() -> ThreadTaskResult?
    // Register a function to run once on every worker thread after the last task completes.
    fn ThreadGroup.set_finalizer_fn(<Task, Result>)(ThreadFinalizerFn thread_finalizer_fn, void* finalizer_context) -> void
    // Register a function to run once on every worker thread before it processes any tasks.
    fn ThreadGroup.set_init_fn(<Task, Result>)(ThreadInitFn thread_init_fn, void* init_context) -> void
    // Set the OS-level thread settings (stack size, priority, etc.) used by start().
    fn ThreadGroup.set_thread_settings(<Task, Result>)(ThreadSettings settings) -> void
    // Startup the threads running this ThreadGroup.
    fn ThreadGroup.start(<Task, Result>)() -> void?
    // Add a task. If the queue is full, return an excuse.
    fn ThreadGroup.try_add_task(<Task, Result>)(Task task) -> void?
    // Pop the task + result, if available, otherwise return with thread::CHANNEL_EMPTY.
    fn ThreadGroup.try_pop_result(<Task, Result>)() -> ThreadTaskResult?
