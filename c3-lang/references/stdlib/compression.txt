## std::compression::deflate
DEFLATE compression implementation (RFC 1951).

 API:
 - fn char[]? decompress(Allocator allocator, char[] input)
 - fn void?   decompress_stream(InStream input, OutStream output)
 - fn char[]? compress(Allocator allocator, char[] input, )
 - struct Inflater: InStream for pull-based decompression (use `init` then `read`).
  // Stateful DEFLATE inflater.

    .dyn_code_huff: Huffman
    .hlit: int
    .hdist: int
    .hclen: int
    .dyn_i: int

    .done: bool
    .state: InflaterState
    fn Inflater.close() -> void?
    fn Inflater.init(InStream input, char[] bit_reader_buf = {}) -> void

## std::compression::gzip
Gzip compression and decompression module (DEFLATE).

 API:
 - fn char[]?   compress(Allocator allocator, char[] src, ...)
 - fn void?     compress_stream(OutStream writer, char[] src, ...)
 - fn Gzip?     uncompress(Allocator allocator, char[] src)
 - fn char[]?   uncompress_bytes(Allocator allocator, char[] src)
 - fn void?     uncompress_stream(OutStream writer, InStream src)
 - struct GzipWriter
 - struct GzipReader

 Initial implementation by Koni Marti.
 Streaming and concatenation support added by Manu Linares.

## std::compression::zip
ZIP archive module (STORE and DEFLATE).

 API:
 - fn ZipArchive? open(Allocator allocator, String path, String mode = "r")
 - fn ZipArchive? recover(Allocator allocator, String path, )
 - fn void?       ZipArchive.extract(&self, String output_dir)
 - fn ZipEntry?   ZipArchive.stat(&self, String filename)
 - fn char[]?     ZipArchive.read_file_all(&self, Allocator allocator, String filename)
 - fn void?       ZipArchive.close(&self)
  // Describes a single entry within a ZIP archive.
  struct ZipEntry
    .name: String
    .uncompressed_size: long
    .compressed_size: long
    .is_directory: bool
    .is_encrypted: bool
    .crc32: uint
    .offset: long
    .method: ZipMethod
    .last_mod_time: ushort
    .last_mod_date: ushort
    fn ZipEntry.time() -> Time
  def ZipEntryList = void
  struct ZipArchive
    .file: File*
    .allocator: Allocator
    .entries: ZipEntryList
    .mode: String
    .path: String
    .comment: String
    // Adds a directory entry to the archive.
    fn ZipArchive.add_directory(String dirname) -> void?
    // Closes the ZIP archive, writing the central directory if in write mode.
    fn ZipArchive.close() -> void?
    fn ZipArchive.count() -> sz
    // Extracts the entire archive to the specified directory.
    fn ZipArchive.extract(String output_dir, String password = "") -> void?
    // Opens a reader for an entry.
    fn ZipArchive.open_reader(String filename, String password = "") -> ZipEntryReader?
    // Opens a writer for a new entry.
    fn ZipArchive.open_writer(String filename, ZipMethod method = DEFLATE) -> ZipEntryWriter?
    // Reads an entire file from the archive.
    fn ZipArchive.read_file_all(Allocator allocator, String filename, String password = "") -> char[]?
    // Returns metadata for the entry with the given filename.
    fn ZipArchive.stat(String filename) -> ZipEntry?
    // Returns metadata for the entry at the given index.
    fn ZipArchive.stat_at(sz index) -> ZipEntry?
    // Writes an entire file to the archive.
    fn ZipArchive.write_file(String filename, char[] data, ZipMethod method = DEFLATE) -> void?
